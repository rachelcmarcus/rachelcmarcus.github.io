What this import pipeline does

This R syntax file takes the raw per-subject event log exports @still-face-data-example (one Excel file per subject) and turns them into ONE wide dataset – one row per subject, one column per scored variable – ready to be further scored in SPSS.

Step by step, it:

  1. Reads every subject’s raw Excel event log.
  2. Checks each file for basic problems (wrong ID, unrecognized behavioral codes).
  3. Finds the Baseline / Still Face / Reunion episode boundaries for that subject (the “anchors”).
  4. Labels every row with which episode it falls in.
  5. Computes every duration/count/joint-attention variable, per episode.
  6. Reshapes each subject from long (one row per coded event) to wide (one row per subject).
  7. Stacks every subject into one dataset and checks for duplicate IDs.
  8. In the source project, saves that one merged, wide dataset as an SPSS .sav file.

2. Import one subject’s raw event log

This section builds the tools to import in a single subject’s raw Excel file and do an initial data quality check before the raw data is scored.

Get the subject ID from a file name

code 
# Every event log file name starts with the subject's numeric ID
# (e.g. "567_SFF_SF6M - Raw Data.xlsx" -> "567")

# This function pulls  ID out of the file name
extract_id_from_filename <- function(filename) {

  # Grab the 3 digits at the very start of the (base, not full-path) file name
  id <- str_extract(basename(filename), "^\\d+")

  # If no leading digits were found, stops rather than continuing with a missing ID
  if (is.na(id)) {
    stop("Could not identify a subject ID from file: ", basename(filename))
  }

  # Returns the extracted ID
  id
}

Import one subject’s file and run basic checks

code 
# This function imports raw data file and conducts initial data cleaning
import_event_log <- function(filename_full) {

  # Keep the short file name around for readable error messages
  filename_short <- basename(filename_full)

  # Get the subject ID using function built above
  filename_id <- extract_id_from_filename(filename_short)

  # Read the raw Excel export
  data <- read_excel(filename_full) %>%
    as.data.frame() # Convert from a tibble to a plain data frame 
  
  # Identify blank template rows: no Behavior coded at all, and both the
  # relative time and duration are exactly 0 -- these are empty leftover
  # rows in the export
  blank_rows <- data %>%
    filter(is.na(Behavior), Time_Relative_sf == 0, Duration_sf == 0)
  
  if (nrow(blank_rows) > 0) {
    # Record this file's blank rows in the running log (<<- writes to the
    # BLANK_ROW_LOG for lead coder to review)
    BLANK_ROW_LOG <<- bind_rows(
      BLANK_ROW_LOG,
      tibble(ID = filename_id, file = filename_short, n_blank_rows = nrow(blank_rows))
    )
    # Drop  blank rows 
    data <- data %>%
      filter(!(is.na(Behavior) & Time_Relative_sf == 0 & Duration_sf == 0))
  }

  # Time_Relative_hms comes in as a full datetime with a fake Excel-created date
  # Reformat to "HH:MM:SS" text 
  data <- data %>%
    mutate(Time_Relative_hms = format(Time_Relative_hms, "%H:%M:%S"))

  # Number every row in a new column called "Code"
  # Creates an event-order count the script uses to find episode boundaries
  data <- data %>%
    tibble::rowid_to_column("Code") %>%
    # Duplicate the "Subject ID" column into a plain, easy-to-type "ID" column
    mutate(ID = as.character(`Subject ID`))  

  # Collect every distinct ID value found inside the file itself
  internal_ids <- unique(data$ID)

  # Data quality check: stop if the file contains more than one Subject ID
  if (length(internal_ids) != 1) {
    stop(
      "File does not contain exactly one Subject ID: ", filename_short,
      ". Found: ", paste(internal_ids, collapse = ", ")
    )
  }

  # Data quality check: Stop if the ID inside the file doesn't match the ID in the file name
  if (internal_ids != filename_id) {
    stop(
      "File name ID (", filename_id, ") does not match internal Subject ID (",
      internal_ids, ") in file: ", filename_short
    )
  }

  # Pull out every Subject/Behavior combination in this file that ISN'T a episode marker 
  observed_behaviors <- data %>%
    filter(!(Behavior %in% EPISODE_MARKERS)) %>%
    distinct(Subject, Behavior)

  # Compare those observed combinations against BEHAVIOR_REFERENCE, keeping only the ones that do NOT have a match there.
  unmapped <- observed_behaviors %>%
    anti_join(BEHAVIOR_REFERENCE, by = c("Subject", "Behavior"))

  # Data quality check: Stop if anything unmapped was found
  if (nrow(unmapped) > 0) {
    stop(
      "Unrecognized Subject/Behavior combination(s) in file ", filename_short,
      ": ", paste(unmapped$Subject, unmapped$Behavior, sep = "/", collapse = "; "),
      ". Add to BEHAVIOR_REFERENCE or fix the coding before re-running."
    )
  }

  # Return the cleaned, validated data for this one subject
  data
}

3. Episode anchors

“Anchors” are the Code numbers marking where each episode (Baseline, Still Face, Reunion) starts and ends for one subject.

Important

Everything after this depends on these episode boundaries being found correctly, so this function stops immediately if an episode marker is missing or the anchors come out of order.

code 
# Function to mark start and stop of each episode
get_episode_anchors <- function(data) {

  # Subject ID, kept for clearer error messages below
  filename_note <- unique(data$ID)

  # Baseline starts at the very first coded row in the file
  bl_start <- min(data$Code)

  # Reunion ends at the very last coded row in the file
  reunion_end <- max(data$Code)

  # "15 S NO INFANT VOCAL" brackets the baseline-to-still-face transition
  # buffer: baseline ends where this marker STARTS, and still face doesn't
  # begin until this marker STOPS. The gap in between (a ~15-30s "do not code"
  # transition) is correctly left unclassified (episode = NA).

  # Every "State start" row for this marker
  no_infant_vocal_start <- data %>%
    filter(Behavior == "15 S NO INFANT VOCAL", Event_Type == "State start")

  # Every "State stop" row for this marker
  no_infant_vocal_stop <- data %>%
    filter(Behavior == "15 S NO INFANT VOCAL", Event_Type == "State stop")

  # Data quality check: Stop if either the start or the stop event is missing
  if (nrow(no_infant_vocal_start) == 0 | nrow(no_infant_vocal_stop) == 0) {
    stop("Could not find both a start and a stop for '15 S NO INFANT VOCAL' marker for ID: ", filename_note)
  }

  # Baseline ends at the first "start" row of the marker
  bl_end <- first(no_infant_vocal_start$Code)

  # Still Face begins at the first "stop" row of the marker
  sf_start <- first(no_infant_vocal_stop$Code)

  # Same start/stop logic for the still-face-to-reunion transition buffer,
  # this time using the "REUNION TRANSITION NO EXPRESSION" marker

  # Every "State start" row for this marker
  reunion_transition_start <- data %>%
    filter(Behavior == "REUNION TRANSITION NO EXPRESSION", Event_Type == "State start")

  # Every "State stop" row for this marker
  reunion_transition_stop <- data %>%
    filter(Behavior == "REUNION TRANSITION NO EXPRESSION", Event_Type == "State stop")

  # Data quality check: Stop if either the start or the stop event is missing
  if (nrow(reunion_transition_start) == 0 | nrow(reunion_transition_stop) == 0) {
    stop("Could not find both a start and a stop for 'REUNION TRANSITION NO EXPRESSION' marker for ID: ", filename_note)
  }

  # Still Face ends at the first "start" row of this marker
  sf_end <- first(reunion_transition_start$Code)

  # Reunion begins at the first "stop" row of this marker
  reunion_start <- first(reunion_transition_stop$Code)

  # Bundle all six anchor points into a one-row tibble, labelled with the subject ID
  anchors <- tibble(
    ID = filename_note,
    bl_start = bl_start, bl_end = bl_end,
    sf_start = sf_start, sf_end = sf_end,
    reunion_start = reunion_start, reunion_end = reunion_end
  )

  # Put the six anchor points in one vector, in the order they SHOULD occur
  anchor_sequence <- c(bl_start, bl_end, sf_start, sf_end, reunion_start, reunion_end)

  # Data quality check: Stop if that sequence is incorrect
  # (e.g. a marker miscoded earlier/later than it should be)
  if (is.unsorted(anchor_sequence, strictly = FALSE)) {
    stop("Episode anchors are not in increasing order for ID: ", filename_note)
  }

  # Return the anchors tibble
  anchors
}

4. Label every row with its episode

episode records which episode each row belongs to: 1 = Baseline, 2= Still Face, 3 = Reunion. Rows that fall in the transition buffers between episodes (including the boundary marker rows themselves) are expected to come out as NA to be correctly excluded from scoring.

code 
# Function to assign each row of data to the correct episode
assign_episode <- function(data, anchors) {
  data %>%
    mutate(episode = case_when(
      # Row falls within the Baseline anchor range
      Code >= anchors$bl_start & Code <= anchors$bl_end ~ 1,
      # Row falls within the Still Face anchor range
      Code >= anchors$sf_start & Code <= anchors$sf_end ~ 2,
      # Row falls within the Reunion anchor range
      Code >= anchors$reunion_start & Code <= anchors$reunion_end ~ 3,
      # Anything else (transition buffers, marker rows) is left as missing
      TRUE ~ NA_real_
    ))
}

5. Put it together: import + anchor + label one subject

code 
# Function to fully import, clean, and label each raw data file
import_and_label_one_subject <- function(filename_full) {

  # Import and validate this subject's raw file
  data <- import_event_log(filename_full)

  # Find this subject's episode boundary "anchors"
  anchors <- get_episode_anchors(data)

  # Add the episode (episode) label to every row
  data <- assign_episode(data, anchors)

  # Keep only the columns scoring actually needs
  data %>%
    select(ID, Code, Subject, Behavior, Event_Type, Time_Relative_sf, Time_Relative_hms, Duration_sf, episode)
}

6. Scoring building blocks

Duration of behavior is only calculated from “State start” rows, as these rows have the full duration of that state, so “State stop” rows are filtered out here to avoid double-counting the same time span.

We then create four small functions that every score in the final dataset is built from: - total duration of a behavior - count of a behavior - overlap between two behaviors (for joint attention) - check for whether an episode had any real coding at allTotal seconds one subject spent in one behavior during one episode.

code 
# Function to compute total duration of behavior in each episodie
compute_duration <- function(data, subj, behavior, episode_num) {
  data %>%
    # Keep only rows matching this subject, this behavior, "State start" events, and this episode
    filter(Subject == subj, Behavior == behavior,
           Event_Type == "State start", episode == episode_num) %>%
    # Add up the duration of every matching row
    summarise(total = sum(Duration_sf)) %>%
    # Pull that single total out as a plain number
    pull(total)
}

# Function for how m any times behavior occurs in an episode
compute_count <- function(data, subj, behavior, episode_num) {
  data %>%
    # Keep only rows matching this subject, this behavior, "State start"
    # events, and this episode -- each such row is one "entry" into the
    # behavior
    filter(Subject == subj, Behavior == behavior,
           Event_Type == "State start", episode == episode_num) %>%
    # Count the matching rows
    nrow()
}

# Function for the overlap (in seconds) between two behaviors to score joint attention (i.e. mom talking/touching WHILE infant looks at mom) 
compute_overlap_duration <- function(data, subject_a, behavior_a, subject_b, behavior_b, episode) {

  # Build every interval (start, end) for behavior A in this episode
  intervals_a <- data %>%
    filter(Subject == subject_a, Behavior == behavior_a,
           Event_Type == "State start", episode == episode) %>%
    # start_a = when it began; end_a = start plus how long it lasted.
    transmute(start_a = Time_Relative_sf, end_a = Time_Relative_sf + Duration_sf)

  # Build every interval (start, end) for behavior B in this episode
  intervals_b <- data %>%
    filter(Subject == subject_b, Behavior == behavior_b,
           Event_Type == "State start", episode == episode) %>%
    # start_b = when it began; end_b = start plus how long it lasted.
    transmute(start_b = Time_Relative_sf, end_b = Time_Relative_sf + Duration_sf)

  # If either behavior never occurred in this episode, there can be no
  # overlap -- returns 0 rather than trying to cross an empty table
  if (nrow(intervals_a) == 0 | nrow(intervals_b) == 0) {
    return(0)
  }

  # Pair every interval of A with every interval of B (a full cross join),
  crossing(intervals_a, intervals_b) %>%
    # then compute how much each pair overlaps: the later of the two
    # starts to the earlier of the two ends, floored at 0 (pmax/pmin) so
    # non-overlapping pairs contribute nothing negative
    mutate(overlap = pmax(0, pmin(end_a, end_b) - pmax(start_a, start_b))) %>%
    # Add up the overlap across every pair,
    summarise(total = sum(overlap)) %>%
    # and pull that single total out as a plain number
    pull(total)
}

# Function to confirm at least one behavior was coded during the episode, a quality check: Distinguishes "episode was coded and at least one behavior happened" resulting in behaviors not coded as 0 from "episode was never coded at all" and then behaviors are coded as NA
episode_has_codes <- function(data, episode_num) {
  data %>%
    # Rows in this episode that are NOT boundary markers
    filter(episode == episode_num, !(Behavior %in% EPISODE_MARKERS)) %>%
    # TRUE if there's at least one such row
    nrow() > 0
}

# Lookup: episode number -> the variable-name prefix used for that episode.
# 6M Still Face - Baseline: v2sfbl; 6M Still Face - Still Face: v2sfsf; 6M Still Face - Reunion: v2sfre; 
EPISODE_PREFIX <- c(`1` = "v2sfbl", `2` = "v2sfsf", `3` = "v2sfre")

# Function: Given a finished one-row "scores" tibble for a subject, blank out every
# v2sf<episode>... column for any episode that had zero real codes (per
# episode_has_codes() above), so an uncoded episode reads as missing rather
# than as all zeros
blank_uncoded_episodes <- function(scores, data) {
  # Check each of the three episodes in turn
  for (episode in c(1, 2, 3)) {
    # Only act on episodes that were never actually coded
    if (!episode_has_codes(data, episode)) {
      # Get this episode's variable-name prefix (e.g. "v2sfsf")
      prefix <- EPISODE_PREFIX[[as.character(episode)]]
      # Find every column in `scores` that starts with that prefix
      cols_to_blank <- names(scores)[str_starts(names(scores), prefix)]
      # Set every one of those columns to NA
      scores[cols_to_blank] <- NA_real_
    }
  }
  # Return the (possibly blanked) scores tibble
  scores
}

# Function to transform data from long to wide 
compute_spec_values_one_subject <- function(data) {
  SCORE_SPEC %>%
    mutate(
      #pmap_dbl() walks the four spec columns (subject, behavior, episode, metric) row by row and returns one number per row
      value = pmap_dbl(
        list(subject, behavior, episode, metric),
        function(subj, behavior, episode, metric) {
          if (metric == "duration") {
            # This row wants a total duration
            compute_duration(data, subj, behavior, episode)
          } else {
            # This row wants a count
            compute_count(data, subj, behavior, episode)
          }
        }
      )
    ) %>%
    # Keep only the variable name and its computed value
    select(var_name, value) %>%
    # Pivot from one-row-per-variable (long) to one-row-total (wide), using var_name as the new column names
    pivot_wider(names_from = var_name, values_from = value)
}

# Function to compute scores, using the functions created above
compute_raw_scores_one_subject <- function(data) {

  # This subject's ID (data has already been filtered to one subject)
  subject_id <- unique(data$ID)

  # Every SCORE_SPEC-driven duration/count value, reshaped wide
  spec_values <- compute_spec_values_one_subject(data)

  # Start a one-row tibble with just the ID,
  scores <- tibble(ID = subject_id) %>%
    # then attach all of the SCORE_SPEC columns next to it
    bind_cols(spec_values)

  # Joint attention: Baseline and Reunion only 
  scores$v2sfbljntatn <- compute_overlap_duration(
    data, "Mother", "Mom Talking/Touching Infant", "Infant", "Looking at Mom", episode = 1
  )
  scores$v2sfrejntatn <- compute_overlap_duration(
    data, "Mother", "Mom Talking/Touching Infant", "Infant", "Looking at Mom", episode = 3
  )

  # If an episode has zero behavior codes, make all behaviors missing (NA)
  scores <- blank_uncoded_episodes(scores, data)

  # Pull subject's episode anchors (start/end code numbers)
  anchors <- get_episode_anchors(data)

# Function to add episode start and end times as new rows in dataset
add_episode_boundary_times <- function(scores, data, anchors) {

  # One row per episode: which number it is, its prefix, and which anchor
  # columns mark its start/end Code numbers.
  episode_info <- tribble(
    ~episode, ~prefix,   ~start_col,       ~end_col,
    1,        "v2sfbl",  "bl_start",       "bl_end",
    2,        "v2sfsf",  "sf_start",       "sf_end",
    3,        "v2sfre",  "reunion_start",  "reunion_end"
  )

  # Walk through the three episodes one at a time
  for (i in seq_len(nrow(episode_info))) {

    # This episode's number (1/2/3)
    episode <- episode_info$episode[i]
    # This episode's variable-name prefix (e.g. "v2sfbl")
    prefix <- episode_info$prefix[i]
    # The Code number where this episode starts, from `anchors`
    start_code <- anchors[[episode_info$start_col[i]]]
    # The Code number where this episode ends, from `anchors`
    end_code <- anchors[[episode_info$end_col[i]]]

    # Look up the literal boundary values directly -- anchors always
    # point to a real row (get_episode_anchors() already stops if a
    # marker is missing), so these are always available regardless of
    # whether any behavior was actually coded inside the episode
    start_hms <- data %>% filter(Code == start_code) %>% pull(Time_Relative_hms)
    end_hms   <- data %>% filter(Code == end_code)   %>% pull(Time_Relative_hms)
    start_sec <- data %>% filter(Code == start_code) %>% pull(Time_Relative_sf)
    end_sec   <- data %>% filter(Code == end_code)   %>% pull(Time_Relative_sf)

    # Blank the boundary times only when the episode is truly empty:
    # start and end land on the exact same moment AND no real (non-marker)
    # behavior was coded anywhere in the episode. An episode with a real,
    # nonzero span but no coded behavior inside it keeps its boundary
    # times -- only its duration/count variables get blanked elsewhere,
    # by blank_uncoded_episodes()
    if (start_sec == end_sec && !episode_has_codes(data, episode)) {
      start_hms <- NA_character_
      end_hms   <- NA_character_
      start_sec <- NA_real_
      end_sec   <- NA_real_
    }

    # Write all four boundary columns into `scores`, using this episode's
    # prefix to build the right column names (e.g. "v2sfblepistarthms")
    scores[[paste0(prefix, "epistarthms")]] <- start_hms
    scores[[paste0(prefix, "epiendhms")]]   <- end_hms
    scores[[paste0(prefix, "epistart")]]    <- start_sec
    scores[[paste0(prefix, "epiend")]]      <- end_sec
    # Episode length in seconds (end minus start; NA automatically if
    # either side is NA).
    scores[[paste0(prefix, "epi")]]         <- end_sec - start_sec
  }

  # Return `scores` with all the boundary-time columns added
  scores
}

# What this import pipeline does

This R syntax file takes the raw per-subject event log exports
@still-face-data-example (one Excel file per subject) and turns them
into ONE wide dataset -- one row per subject, one column per scored
variable -- ready to be further scored in SPSS.

Step by step, it:

1.  Reads every subject's raw Excel event log.
2.  Checks each file for basic problems (wrong ID, unrecognized
    behavioral codes).
3.  Finds the Baseline / Still Face / Reunion episode boundaries for
    that subject (the "anchors").
4.  Labels every row with which episode it falls in.
5.  Computes every duration/count/joint-attention variable, per episode.
6.  Reshapes each subject from long (one row per coded event) to wide
    (one row per subject).
7.  Stacks every subject into one dataset and checks for duplicate IDs.
8.  In the source project, saves that one merged, wide dataset as an
    SPSS `.sav` file.

```{r}
  filename_short <- basename(filename_full)

  # Get the subject ID using function built above
  filename_id <- extract_id_from_filename(filename_short)

  # Read the raw Excel export
  data <- read_excel(filename_full) %>%
    as.data.frame() # Convert from a tibble to a plain data frame

  # relative time and duration are exactly 0 -- these are empty leftover
  # rows in the export
  blank_rows <- data %>%
    filter(is.na(Behavior), Time_Relative_sf == 0, Duration_sf == 0)
  
  if (nrow(blank_rows) > 0) {
    # Record this file's blank rows in the running log (<<- writes to the
    # BLANK_ROW_LOG for lead coder to review)
    BLANK_ROW_LOG <<- bind_rows(
      BLANK_ROW_LOG,
      tibble(ID = filename_id, file = filename_short, n_blank_rows = nrow(blank_rows))
    )
    # Drop  blank rows 
    data <- data %>%
      filter(!(is.na(Behavior) & Time_Relative_sf == 0 & Duration_sf == 0))
  }

  # Reformat to "HH:MM:SS" text 
  data <- data %>%
    mutate(Time_Relative_hms = format(Time_Relative_hms, "%H:%M:%S"))

  # Number every row in a new column called "Code"
  # Creates an event-order count the script uses to find episode boundaries
  data <- data %>%
    tibble::rowid_to_column("Code") %>%
    # Duplicate the "Subject ID" column into a plain, easy-to-type "ID" column
    mutate(ID = as.character(`Subject ID`))  

  # Collect every distinct ID value found inside the file itself
  internal_ids <- unique(data$ID)

  # Data quality check: stop if the file contains more than one Subject ID
  if (length(internal_ids) != 1) {
    stop(
      "File does not contain exactly one Subject ID: ", filename_short,
      ". Found: ", paste(internal_ids, collapse = ", ")
    )
  }

  # Data quality check: Stop if the ID inside the file doesn't match the ID in the file name
  if (internal_ids != filename_id) {
    stop(
      "File name ID (", filename_id, ") does not match internal Subject ID (",
      internal_ids, ") in file: ", filename_short
    )
  }

  # Pull out every Subject/Behavior combination in this file that ISN'T a episode marker 
  observed_behaviors <- data %>%
    filter(!(Behavior %in% EPISODE_MARKERS)) %>%
    distinct(Subject, Behavior)

  # Compare those observed combinations against BEHAVIOR_REFERENCE, keeping only the ones that do NOT have a match there.
  unmapped <- observed_behaviors %>%
    anti_join(BEHAVIOR_REFERENCE, by = c("Subject", "Behavior"))

  # Data quality check: Stop if anything unmapped was found
  if (nrow(unmapped) > 0) {

    )
  }

  # Return the cleaned, validated data for this one subject
  data
```

# 3. Episode anchors

"Anchors" are the `Code` numbers marking where each episode (Baseline,
Still Face, Reunion) starts and ends for one subject.

::: callout-important
Everything after this depends on these episode boundaries being found
correctly, so this function stops immediately if an episode marker is
missing or the anchors come out of order.
:::

```{r}
#| label: episode-anchor-function
# Function to mark start and stop of each episode
get_episode_anchors <- function(data) {

  # Subject ID, kept for clearer error messages below
  filename_note <- unique(data$ID)

  # Baseline starts at the very first coded row in the file
  bl_start <- min(data$Code)

  # Reunion ends at the very last coded row in the file
  reunion_end <- max(data$Code)

  # "15 S NO INFANT VOCAL" brackets the baseline-to-still-face transition
  # buffer: baseline ends where this marker STARTS, and still face doesn't
  # begin until this marker STOPS. The gap in between (a ~15-30s "do not code"
  # transition) is correctly left unclassified (episode = NA).

  # Every "State start" row for this marker
  no_infant_vocal_start <- data %>%
    filter(Behavior == "15 S NO INFANT VOCAL", Event_Type == "State start")

  # Every "State stop" row for this marker
  no_infant_vocal_stop <- data %>%
    filter(Behavior == "15 S NO INFANT VOCAL", Event_Type == "State stop")

  # Data quality check: Stop if either the start or the stop event is missing
  if (nrow(no_infant_vocal_start) == 0 | nrow(no_infant_vocal_stop) == 0) {
    stop("Could not find both a start and a stop for '15 S NO INFANT VOCAL' marker for ID: ", filename_note)
  }

  # Baseline ends at the first "start" row of the marker
  bl_end <- first(no_infant_vocal_start$Code)

  # Still Face begins at the first "stop" row of the marker
  sf_start <- first(no_infant_vocal_stop$Code)

  # Same start/stop logic for the still-face-to-reunion transition buffer,
  # this time using the "REUNION TRANSITION NO EXPRESSION" marker

  # Every "State start" row for this marker
  reunion_transition_start <- data %>%
    filter(Behavior == "REUNION TRANSITION NO EXPRESSION", Event_Type == "State start")

  # Every "State stop" row for this marker
  reunion_transition_stop <- data %>%
    filter(Behavior == "REUNION TRANSITION NO EXPRESSION", Event_Type == "State stop")

  # Data quality check: Stop if either the start or the stop event is missing
  if (nrow(reunion_transition_start) == 0 | nrow(reunion_transition_stop) == 0) {
    stop("Could not find both a start and a stop for 'REUNION TRANSITION NO EXPRESSION' marker for ID: ", filename_note)
  }

  # Still Face ends at the first "start" row of this marker
  sf_end <- first(reunion_transition_start$Code)

  # Reunion begins at the first "stop" row of this marker
  reunion_start <- first(reunion_transition_stop$Code)

  # Bundle all six anchor points into a one-row tibble, labelled with the subject ID
  anchors <- tibble(
    ID = filename_note,
    bl_start = bl_start, bl_end = bl_end,
    sf_start = sf_start, sf_end = sf_end,
    reunion_start = reunion_start, reunion_end = reunion_end
  )

  # Put the six anchor points in one vector, in the order they SHOULD occur
  anchor_sequence <- c(bl_start, bl_end, sf_start, sf_end, reunion_start, reunion_end)

  # Data quality check: Stop if that sequence is incorrect
  # (e.g. a marker miscoded earlier/later than it should be)
  if (is.unsorted(anchor_sequence, strictly = FALSE)) {
    stop("Episode anchors are not in increasing order for ID: ", filename_note)
  }

  # Return the anchors tibble
  anchors
}
```

# 4. Label every row with its episode

`episode` records which episode each row belongs to: `1` = Baseline,
`2`= Still Face, `3` = Reunion. Rows that fall in the transition buffers
between episodes (including the boundary marker rows themselves) are
expected to come out as `NA` to be correctly excluded from scoring.

```{r}
#| label: episode-function
# Function to assign each row of data to the correct episode
assign_episode <- function(data, anchors) {
  data %>%
    mutate(episode = case_when(
      # Row falls within the Baseline anchor range
      Code >= anchors$bl_start & Code <= anchors$bl_end ~ 1,
      # Row falls within the Still Face anchor range
      Code >= anchors$sf_start & Code <= anchors$sf_end ~ 2,
      # Row falls within the Reunion anchor range
      Code >= anchors$reunion_start & Code <= anchors$reunion_end ~ 3,
      # Anything else (transition buffers, marker rows) is left as missing
      TRUE ~ NA_real_
    ))
}
```

# 6. Scoring building blocks

Duration of behavior is only calculated from "State start" rows, as
these rows have the full duration of that state, so "State stop" rows
are filtered out here to avoid double-counting the same time span.

We then create four small functions that every score in the final
dataset is built from: - total duration of a behavior - count of a
behavior - overlap between two behaviors (for joint attention) - check
for whether an episode had any real coding at allTotal seconds one
subject spent in one behavior during one episode.

```{r}
#| label: duration-function
# Function to compute total duration of behavior in each episodie
compute_duration <- function(data, subj, behavior, episode_num) {
  data %>%
    # Keep only rows matching this subject, this behavior, "State start" events, and this episode
    filter(Subject == subj, Behavior == behavior,
           Event_Type == "State start", episode == episode_num) %>%
    # Add up the duration of every matching row
    summarise(total = sum(Duration_sf)) %>%
    # Pull that single total out as a plain number
    pull(total)
}
```

```{r}
#| label: overlap-function
# Function for the overlap (in seconds) between two behaviors to score joint attention (i.e. mom talking/touching WHILE infant looks at mom) 
compute_overlap_duration <- function(data, subject_a, behavior_a, subject_b, behavior_b, episode) {

  # Build every interval (start, end) for behavior A in this episode
  intervals_a <- data %>%
    filter(Subject == subject_a, Behavior == behavior_a,
           Event_Type == "State start", episode == episode) %>%
    # start_a = when it began; end_a = start plus how long it lasted.
    transmute(start_a = Time_Relative_sf, end_a = Time_Relative_sf + Duration_sf)

  # Build every interval (start, end) for behavior B in this episode
  intervals_b <- data %>%
    filter(Subject == subject_b, Behavior == behavior_b,
           Event_Type == "State start", episode == episode) %>%
    # start_b = when it began; end_b = start plus how long it lasted.
    transmute(start_b = Time_Relative_sf, end_b = Time_Relative_sf + Duration_sf)

  # If either behavior never occurred in this episode, there can be no
  # overlap -- returns 0 rather than trying to cross an empty table
  if (nrow(intervals_a) == 0 | nrow(intervals_b) == 0) {
    return(0)
  }

  # Pair every interval of A with every interval of B (a full cross join),
  crossing(intervals_a, intervals_b) %>%
    # then compute how much each pair overlaps: the later of the two
    # starts to the earlier of the two ends, floored at 0 (pmax/pmin) so
    # non-overlapping pairs contribute nothing negative
    mutate(overlap = pmax(0, pmin(end_a, end_b) - pmax(start_a, start_b))) %>%
    # Add up the overlap across every pair,
    summarise(total = sum(overlap)) %>%
    # and pull that single total out as a plain number
    pull(total)
}
```

```{r}
# Function: Given a finished one-row "scores" tibble for a subject, blank out every

blank_uncoded_episodes <- function(scores, data) {

  scores
```

```{r}
#| label: compute-spec-values
# Function to transform data from long to wide 
compute_spec_values_one_subject <- function(data) {
  SCORE_SPEC %>%
    mutate(
      #pmap_dbl() walks the four spec columns (subject, behavior, episode, metric) row by row and returns one number per row
      value = pmap_dbl(
        list(subject, behavior, episode, metric),
        function(subj, behavior, episode, metric) {
          if (metric == "duration") {
            # This row wants a total duration
            compute_duration(data, subj, behavior, episode)
          } else {
            # This row wants a count
            compute_count(data, subj, behavior, episode)
          }
        }
      )
    ) %>%
    # Keep only the variable name and its computed value
    select(var_name, value) %>%
    # Pivot from one-row-per-variable (long) to one-row-total (wide), using var_name as the new column names
    pivot_wider(names_from = var_name, values_from = value)
}
```

```{r}
#| label: episode-boundary-times-function
# Function to add episode start and end times as new rows in dataset
add_episode_boundary_times <- function(scores, data, anchors) {

  # One row per episode: which number it is, its prefix, and which anchor
  # columns mark its start/end Code numbers.
  episode_info <- tribble(
    ~episode, ~prefix,   ~start_col,       ~end_col,
    1,        "v2sfbl",  "bl_start",       "bl_end",
    2,        "v2sfsf",  "sf_start",       "sf_end",
    3,        "v2sfre",  "reunion_start",  "reunion_end"
  )

  # Walk through the three episodes one at a time
  for (i in seq_len(nrow(episode_info))) {

    # This episode's number (1/2/3)
    episode <- episode_info$episode[i]
    # This episode's variable-name prefix (e.g. "v2sfbl")
    prefix <- episode_info$prefix[i]
    # The Code number where this episode starts, from `anchors`
    start_code <- anchors[[episode_info$start_col[i]]]
    # The Code number where this episode ends, from `anchors`
    end_code <- anchors[[episode_info$end_col[i]]]

    # Look up the literal boundary values directly -- anchors always
    # point to a real row (get_episode_anchors() already stops if a
    # marker is missing), so these are always available regardless of
    # whether any behavior was actually coded inside the episode
    start_hms <- data %>% filter(Code == start_code) %>% pull(Time_Relative_hms)
    end_hms   <- data %>% filter(Code == end_code)   %>% pull(Time_Relative_hms)
    start_sec <- data %>% filter(Code == start_code) %>% pull(Time_Relative_sf)
    end_sec   <- data %>% filter(Code == end_code)   %>% pull(Time_Relative_sf)

    # Blank the boundary times only when the episode is truly empty:
    # start and end land on the exact same moment AND no real (non-marker)
    # behavior was coded anywhere in the episode. An episode with a real,
    # nonzero span but no coded behavior inside it keeps its boundary
    # times -- only its duration/count variables get blanked elsewhere,
    # by blank_uncoded_episodes()
    if (start_sec == end_sec && !episode_has_codes(data, episode)) {
      start_hms <- NA_character_
      end_hms   <- NA_character_
      start_sec <- NA_real_
      end_sec   <- NA_real_
    }

    # Write all four boundary columns into `scores`, using this episode's
    # prefix to build the right column names (e.g. "v2sfblepistarthms")
    scores[[paste0(prefix, "epistarthms")]] <- start_hms
    scores[[paste0(prefix, "epiendhms")]]   <- end_hms
    scores[[paste0(prefix, "epistart")]]    <- start_sec
    scores[[paste0(prefix, "epiend")]]      <- end_sec
    # Episode length in seconds (end minus start; NA automatically if
    # either side is NA).
    scores[[paste0(prefix, "epi")]]         <- end_sec - start_sec
  }

  # Return `scores` with all the boundary-time columns added
  scores
}
```

```{r}
# Convert time variables to the same format 
to_seconds <- function(x) {
  parts <- str_split(str_trim(x), ":")
  map_dbl(parts, function(p) {
    p <- as.numeric(p)
    if (length(p) == 2) {
      p[1] * 60 + p[2]                    # M:SS
    } else if (length(p) == 3) {
      p[1] * 3600 + p[2] * 60 + p[3]       # H:MM:SS
    } else {
      NA_real_
    }
  })
}
```
